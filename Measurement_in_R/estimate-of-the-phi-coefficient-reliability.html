Chapter 12 Estimate of the phi-coefficient reliability

Err_Abs<- VCr/n_r + VCi/n_i + VCpr/n_r + VCpi/n_i + VCri/(n_r*n_i) + VCpri_e/(n_r*n_i)
Phi<- VCp/(VCp + Err_Abs)
round(Phi,3)
## [1] 0.847

Again, we can project with variations on the number of items and raters, using a custom function:

d_studier <- function(D_r, D_i){
   Err_Abs<- VCr/D_r  + VCi/D_i  + VCpr/D_r + VCpi/D_i + VCri/(D_r*D_i) + VCpri_e/(D_r*D_i)
   Phi    <- round( VCp/(VCp + Err_Abs), 3)
              Dstudy <- data.frame(N.Item    = D_i, 
                        N.Rater   = D_r, 
                        "Phi-coef"= Phi)
}
# Let's make some n-sizes for items and raters.
n_item  <- c(1, 3, 5, 7)
n_raters<- rep(c(1,3,5),each = length(n_item))

Dstudy  <- d_studier(D_r = n_raters, D_i= n_item)
Dstudy
